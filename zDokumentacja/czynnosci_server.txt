*** Konfiguracja nginx ***

sudo nano /etc/nginx/sites-available/powerguard.cloud

#Wkleić:
---
# HTTP -> HTTPS for powerguard.cloud
server {
    listen 80;
    listen [::]:80;
    server_name powerguard.cloud www.powerguard.cloud;

    return 301 https://powerguard.cloud$request_uri;
}

# HTTPS
server {
    listen 443 ssl;
    listen [::]:443 ssl;
    http2 on;

    server_name powerguard.cloud;

    ssl_certificate     /etc/letsencrypt/live/powerguard.cloud/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/powerguard.cloud/privkey.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    proxy_set_header Host              $host;
    proxy_set_header X-Real-IP         $remote_addr;
    proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";

    location /powerguard-bcknd/ {
        proxy_pass http://127.0.0.1:9084;
    }

    location / {
        proxy_pass http://127.0.0.1:8084;
    }
}
===

#Enable the site
sudo ln -s /etc/nginx/sites-available/powerguard.cloud /etc/nginx/sites-enabled/powerguard.cloud

#Check
sudo nginx -t

#Reload
sudo systemctl reload nginx

#Test
curl -I http://powerguard.cloud


#Utworzenie certyfikatu
sudo certbot --nginx -d powerguard.cloud -d www.powerguard.cloud

#Check
sudo nginx -t

#Reload
sudo systemctl reload nginx
